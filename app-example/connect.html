TS lib/supabase.ts 
import { createClient } from '@supabase/supabase-js'

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!
const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!

export const supabase = createClient(supabaseUrl, supabaseAnonKey)

// Usage example — fetch data:
const { data, error } = await supabase
  .from('your_table')
  .select('*')

// Auth example:
await supabase.auth.signInWithPassword({
  email: 'user@example.com',
  password: 'password'
})

PY supabase_client.py 
# pip install supabase
from supabase import create_client

url = "https://api.aaryavtech.online"
key = "YOUR_ANON_KEY"  # or service_role for server

supabase = create_client(url, key)

# Fetch data
response = supabase.table("drivers").select("*").execute()
print(response.data)

# Insert data
response = supabase.table("drivers").insert({
    "name": "Chirag",
    "status": "available"
}).execute()
DART main.dart 
// pubspec.yaml: supabase_flutter: ^2.0.0
import 'package:supabase_flutter/supabase_flutter.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  await Supabase.initialize(
    url: 'https://api.aaryavtech.online',
    anonKey: 'YOUR_ANON_KEY',
  );

  runApp(MyApp());
}

// Access client anywhere:
final supabase = Supabase.instance.client;
